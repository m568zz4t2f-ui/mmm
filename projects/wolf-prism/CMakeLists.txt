cmake_minimum_required(VERSION 3.22)
project(WolfPrism VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(prism_dsp_tests tests/dsp_tests.cpp)
target_include_directories(prism_dsp_tests PRIVATE Source)
add_test(NAME dsp COMMAND prism_dsp_tests)
option(PRISM_DSP_ONLY "Build only DSP checks" OFF)
if(PRISM_DSP_ONLY)
 return()
endif()
include(FetchContent)
if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_JUCE)
 set(prism_vendor "${CMAKE_CURRENT_BINARY_DIR}/vendor-juce")
 if(NOT EXISTS "${prism_vendor}/CMakeLists.txt")
  file(MAKE_DIRECTORY "${prism_vendor}")
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE-8.0.12.tar.gz" DESTINATION "${prism_vendor}")
 endif()
 set(FETCHCONTENT_SOURCE_DIR_JUCE "${prism_vendor}")
endif()
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
FetchContent_MakeAvailable(JUCE)
set(prism_formats VST3 Standalone)
if(APPLE)
 list(APPEND prism_formats AU)
endif()
juce_add_plugin(WolfPrism COMPANY_NAME "Wolf Audio" BUNDLE_ID "audio.wolf.prism" PLUGIN_MANUFACTURER_CODE Wolf PLUGIN_CODE Wprm PRODUCT_NAME "Wolf Prism" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Wolf Prism processes your selected audio input." VST3_CATEGORIES Fx Modulation Delay Reverb Distortion AU_MAIN_TYPE kAudioUnitType_MusicEffect FORMATS ${prism_formats})
target_sources(WolfPrism PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(WolfPrism PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(WolfPrism PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
add_executable(prism_host_tests tests/host_tests.cpp)
target_link_libraries(prism_host_tests PRIVATE WolfPrism)
get_target_property(prism_includes juce::juce_audio_processors INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(prism_host_tests PRIVATE ${prism_includes})
target_compile_definitions(prism_host_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
add_test(NAME host COMMAND prism_host_tests)
add_executable(prism_music_render tests/music_render.cpp)
target_link_libraries(prism_music_render PRIVATE WolfPrism)
target_include_directories(prism_music_render PRIVATE ${prism_includes})
target_compile_definitions(prism_music_render PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
