include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_JUCE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE-8.0.12.tar.gz")
 set(tide_vendor "${CMAKE_CURRENT_BINARY_DIR}/vendor-juce")
 if(NOT EXISTS "${tide_vendor}/CMakeLists.txt")
  file(MAKE_DIRECTORY "${tide_vendor}")
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE-8.0.12.tar.gz" DESTINATION "${tide_vendor}")
 endif()
 set(FETCHCONTENT_SOURCE_DIR_JUCE "${tide_vendor}")
endif()
FetchContent_MakeAvailable(JUCE)
set(tide_formats VST3 Standalone)
if(APPLE)
 list(APPEND tide_formats AU)
endif()
juce_add_plugin(WolfTide COMPANY_NAME "Wolf Audio" BUNDLE_ID "audio.wolf.tide"
 PLUGIN_MANUFACTURER_CODE Wolf PLUGIN_CODE Wtid PRODUCT_NAME "Wolf Tide"
 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
 MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Wolf Tide processes live audio from your selected input device."
 VST3_CATEGORIES Fx Delay Reverb AU_MAIN_TYPE kAudioUnitType_Effect FORMATS ${tide_formats})
# Build identity follows the compiled source, including local changes.
set(tide_source_hashes "")
foreach(tide_source Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h Source/TideLook.h Source/PresetBrowser.h Source/Parameters.h Source/ModernParameters.h Source/ModernDSP.h Source/VersionedDSP.h Source/ModernPanel.h Source/TideDSP.h Source/BloomDSP.h)
 file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${tide_source}" tide_hash)
 string(APPEND tide_source_hashes "${tide_hash}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${tide_source}")
endforeach()
string(SHA256 tide_build_hash "${tide_source_hashes}")
string(SUBSTRING "${tide_build_hash}" 0 12 tide_build_id)
configure_file(Source/BuildInfo.h.in "${CMAKE_CURRENT_BINARY_DIR}/TideBuild.h" @ONLY)
target_include_directories(WolfTide PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
target_sources(WolfTide PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(WolfTide PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(WolfTide PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
add_executable(tide_host_tests tests/host_tests.cpp)
target_link_libraries(tide_host_tests PRIVATE WolfTide)
get_target_property(tide_includes juce::juce_audio_processors INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(tide_host_tests PRIVATE ${tide_includes})
target_compile_definitions(tide_host_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
